\noindent
\textbf{Given:}
\begin{itemize}
  \item Copper material with $E = \SI{18e3}{ksi}$.
  \item Diameters: $\varnothing_{AB} = 0.75''$, $\varnothing_{BC} = 1''$, $\varnothing_{CD} = 0.5''$.
\end{itemize}
\textbf{Find:} $\delta_{AD}$.

\[
  \delta_{AD} = \delta_{AB} + \delta_{BC} + \delta_{CD}
  = \frac{N_{AB}L_{AB}}{E_{AB}A_{AB}} + \frac{N_{BC}L_{BC}}{E_{BC}A_{BC}}
    + \frac{N_{CD}L_{CD}}{E_{CD}A_{CD}}
\]

\noindent
\begin{minipage}[c]{0.5\textwidth}
\textbf{``Cut'' in $AB$}\\[0.3em]
\centering
\begin{tikzpicture}
  \cutbar{0}{1.4}{0.2}{2}
  \draw[force] (-1.3,0) node[left] {$\SI{8}{kip}$} -- (-0.05,0);

\end{minipage}%

\begin{minipage}[c]{0.45\textwidth}

\end{gather*}
\end{minipage}

\begin{minipage}[c]{0.45\textwidth}
\begin{gather*}
  \textstyle\sum F_x = 0 \;\rightarrow\; -N_{CD} + \SI{6}{kip} = 0 \\
  \boxed{N_{CD} = \SI{6}{kip}}
\end{gather*}
\end{minipage}

\begin{align*}
  \delta_{AB} &= \frac{(\SI{-8e3}{lb})(80'')}{(\SI{18e6}{psi})\left(\frac{\pi}{4}(0.75'')^2\right)}
               = -0.080'' \\[4pt]
  \delta_{BC} &= \frac{(\SI{2e3}{lb})(150'')}{(\SI{18e6}{psi})\left(\frac{\pi}{4}(1'')^2\right)}
               = 0.021'' \\[4pt]
  \delta_{CD} &= \frac{(\SI{6e3}{lb})(100'')}{(\SI{18e6}{psi})\left(\frac{\pi}{4}(0.5'')^2\right)}
               = 0.170''
\end{align*}
\[
  \boxed{\delta_{AD} = \delta_{AB} + \delta_{BC} + \delta_{CD} = 0.111''}
\]

\section*{Distributed Load Example}

\noindent
\begin{minipage}[c]{0.5\textwidth}
\centering
\begin{tikzpicture}

    \draw[-{Stealth[length=2mm,width=1.6mm]}, semithick] (\x,0) -- (\x+0.65,0);
  \node[pin] at (0,0) {}; \node[left] at (-0.1,0) {$A$};
  \node[right] at (5.35,0) {$B$};
  \draw[thinarrow] (2.3,1.1) -- (2.0,0.1);
  \node[above] at (2.3,1.1) {$w(x)$};
  \draw[dimto] (0,-0.7) -- (1,-0.7) node[right] {$x$};
  \draw[dim] (0,-1.3) -- (5,-1.3) node[midway, fill=white] {$L$};
\end{tikzpicture}
\end{minipage}%
\hfill
\begin{minipage}[c]{0.45\textwidth}
\begin{itemize}
  \item $w = 2$~kips/in., $L = 100''$
  \item Steel rod ($E = \SI{29e3}{ksi}$)
  \item Area: \SI{1}{in^2}
\end{itemize}
\textbf{Find:} $\delta_{AB}$. We will use
\[ \delta = \int_0^L \frac{N(x)}{E(x)A(x)}\,dx. \]
\end{minipage}

\subsection*{``Cut'' at position $x$}

\noindent
\begin{minipage}[c]{0.5\textwidth}
\centering
\begin{tikzpicture}
  \cutbar{0}{2.4}{0.3}{2}
  \foreach \x in {0.15,0.9,1.65}
    \draw[-{Stealth[length=2mm,width=1.6mm]}, semithick] (\x,0) -- (\x+0.55,0);
  \node[pin] at (0,0) {}; \node[left] at (-0.1,0) {$A$};
  \node[above] at (1.2,0.3) {$w$};
  \draw[force] (2.5,0) -- (3.4,0) node[right] {$N(x)$};
  \draw[dim] (0,-0.7) -- (2.4,-0.7) node[midway, fill=white] {$x$};
\end{tikzpicture}

\vspace{0.6em}
Equivalent load:\\[0.3em]
\begin{tikzpicture}
  \cutbar{0}{2.4}{0.3}{2}
  \node[pin] at (1.2,0) {};
  \draw[force] (-1.0,0) node[left] {$wx$} -- (1.15,0);
  \draw[force] (2.5,0) -- (3.4,0) node[right] {$N(x)$};
  \draw[dim] (0,-0.7) -- (2.4,-0.7) node[midway, fill=white] {$x$};
\end{tikzpicture}
\end{minipage}%
\hfill
\begin{minipage}[c]{0.45\textwidth}
\begin{gather*}
  \textstyle\sum F_x = 0 \\
  wx + N(x) = 0 \\
  \boxed{N(x) = -wx}
\end{gather*}
\end{minipage}

\begin{align*}
  \delta_{AB} &= \int_0^L \frac{N(x)}{EA}\,dx
               = \int_0^L \frac{-wx}{EA}\,dx
               = -\frac{w}{EA}\int_0^L x\,dx
               = -\frac{w}{EA}\left(\tfrac12 x^2\right)\Big|_0^L
\end{align*}
\[
  \boxed{\delta_{AB} = -\frac{wL^2}{2EA} = -0.35''}
\]
