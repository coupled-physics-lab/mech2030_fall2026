\section*{3) Stress}

\noindent
\begin{minipage}[c]{0.55\textwidth}
\begin{align*}
  \sigma_{\max} &= \frac{M_{\max}\,c}{I}
   = \frac{M_{\max}\left(\frac{a}{2}\right)}{\frac{a^4}{12}} \\[4pt]
  \sigma_{\max} &= \frac{6M_{\max}}{a^3}
\end{align*}
\end{minipage}%
\hfill
\begin{minipage}[c]{0.42\textwidth}
\centering
\begin{tikzpicture}
  \draw[thick, fill=gray!12] (0,-0.8) rectangle (1.6,0.8);
  \draw[thin, dashed] (-0.3,0) -- (1.9,0) node[right] {N.A.};
  \draw[dim] (-0.35,0) -- (-0.35,0.8) node[midway, left] {$\frac{a}{2}$};
  \draw[axis] (-1.2,0.1) -- (-1.2,0.9) node[above] {$y$};
\end{tikzpicture}
\end{minipage}

\bigskip
\noindent
$\rightarrow$ Solve for $a$, setting $\sigma_{\max} = \sigma_{\text{allow}}$:
\begin{align*}
  a^3 &= \frac{6M_{\max}}{\sigma_{\text{allow}}} \\[4pt]
  a &= \left(\frac{6M_{\max}}{\sigma_{\text{allow}}}\right)^{1/3}
     = \left(\frac{6\,(\SI{2e3}{N.m})}{\SI{50e6}{Pa}}\right)^{1/3}
\end{align*}
\[
  \boxed{a = \SI{0.0621}{m} = \SI{62.1}{mm}}
\]
